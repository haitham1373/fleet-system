# دالة لتحويل الصورة إلى Base64
def get_image_base64(path):
    if path and isinstance(path, str) and os.path.exists(path):
        with open(path, "rb") as f:
            return base64.b64encode(f.read()).decode()
    return ""

def esc(v):
    return html.escape("" if v is None else str(v))

PBKDF2_ITERS = 200_000

def hash_password(password):
    salt = secrets.token_hex(16)
    h = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), bytes.fromhex(salt), PBKDF2_ITERS).hex()
    return f"pbkdf2${PBKDF2_ITERS}${salt}${h}"

def verify_password(password, stored):
    if not stored or not str(stored).startswith("pbkdf2$"):
        return False
    try:
        _, iters, salt, h = str(stored).split("$")
        calc = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), bytes.fromhex(salt), int(iters)).hex()
        return hmac.compare_digest(calc, h)
    except Exception:
        return False

def save_upload(folder, prefix, uploaded):
    orig = os.path.basename(str(uploaded.name))
    stem, ext = os.path.splitext(orig)
    ext = ext.lower() if ext.lower() in (".pdf", ".png", ".jpg", ".jpeg") else ""
    clean = lambda s: (re.sub(r"[^\w\-]+", "_", str(s), flags=re.UNICODE).strip("_")[:40] or "file")
    name = f"{clean(prefix)}_{datetime.now().strftime('%Y%m%d_%H%M%S_%f')}_{clean(stem)}{ext}"
    path = os.path.join(folder, name)
    with open(path, "wb") as f:
        f.write(uploaded.getbuffer())
    return path
